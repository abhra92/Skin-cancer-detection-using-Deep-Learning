\section*{General Formatting Guidelines for a Research Paper}

\begin{itemize}
  \item \textbf{Font:} Times New Roman 12, Arial 11, or Calibri 11
  \item \textbf{Spacing:} Double-spaced
  \item \textbf{Margins:} 1 inch on all sides
  \item \textbf{Indentation:} First line of each paragraph
  \item \textbf{Page Numbers:} Top-right corner
\end{itemize}
